import { navigate } from "@/lib/router/navigate";
import { useNavigate } from "raviger";

export function Hook() {
  const navigate = useNavigate();
  return <button onClick={() => navigate("/licenses")}>Hook</button>;
}

export function Prop({ navigate }: { navigate: (url: string) => void }) {
  return <button onClick={() => navigate("/licenses")}>Prop</button>;
}

export function Module() {
  return <button onClick={() => navigate({ to: "/licenses" })}>Module</button>;
}
