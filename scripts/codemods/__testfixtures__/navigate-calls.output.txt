import { navigate } from "@/lib/router/navigate";
import { navigate as ravigerNavigate } from "raviger";

export function go(facilityId: string, id: string, target: string) {
  navigate({ to: "/licenses" });
  navigate({ to: "/facility/$facilityId/settings/devices/$id", params: { facilityId, id } });
  navigate(
    { to: "/facility/$facilityId/settings", params: { facilityId }, replace: true }
  );
  navigate(
    { to: "/facility/$facilityId/settings/devices", params: { facilityId }, replace: false }
  );
  ravigerNavigate("/does/not/exist");
  ravigerNavigate(`/facility/${facilityId}/settings/devices`, { query: { a: "b" } });
  ravigerNavigate(`/facility/${facilityId}/settings/devices`, { state: { a: 1 } });
  ravigerNavigate(`/facility/${facilityId}/settings/devices`, { replace: true, state: {} });
  ravigerNavigate(`/facility/${facilityId}/settings/devices`, { replace });
  ravigerNavigate(`/facility/${facilityId}/settings/devices`, options);
  ravigerNavigate(-1);
  ravigerNavigate(target);
  ravigerNavigate(`${target}/edit`);
  ravigerNavigate("devices/create");
  ravigerNavigate();
}
