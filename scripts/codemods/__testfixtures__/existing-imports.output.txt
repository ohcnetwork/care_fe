import { type LinkProps, useRouter, Link } from "@tanstack/react-router";
import { navigate as typedNavigate } from "@/lib/router/navigate";

export function Existing({ facilityId }: { facilityId: string }) {
  const router = useRouter();
  return (
    <Link
      to="/facility/$facilityId/settings"
      params={{ facilityId }}
      onClick={() => typedNavigate({ to: "/login" })}>
      {String(router)}
    </Link>
  );
}
