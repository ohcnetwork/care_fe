import { Link } from "@tanstack/react-router";
import { navigate } from "@/lib/router/navigate";
import { Link as RavigerLink } from "raviger";

export function Literal({ id }: { id: string }) {
  navigate({ to: "/patient/$id/$tab", params: { id, tab: "updates" } });
  return (
    <div>
      <Link to="/patient/$id/$tab" params={{ id, tab: "updates" }}>Updates</Link>
      <Link to="/facility/$facilityId/settings" params={{ facilityId: "abc" }}>Fixed facility</Link>
      <RavigerLink href={`/patient/${id}/up dates`}>Not plain</RavigerLink>
    </div>
  );
}
