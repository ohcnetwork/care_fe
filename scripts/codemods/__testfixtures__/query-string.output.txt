import { Link } from "@tanstack/react-router";
import { navigate } from "@/lib/router/navigate";
import { Link as RavigerLink } from "raviger";

export function Query({ facilityId, filter }: { facilityId: string; filter: string }) {
  return (
    <div>
      <Link to="/login" search={{ next: "home" }}>Login</Link>
      <Link
        to="/facility/$facilityId/settings/devices"
        params={{ facilityId }}
        search={{ type: "bed", page: "2" }}>Beds</Link>
      <RavigerLink href={`/facility/${facilityId}/settings/devices?type=${filter}`}>Dynamic</RavigerLink>
      <RavigerLink href="/login?a=b&a=c">Duplicate key</RavigerLink>
      <RavigerLink href="/login?q=hello%20world">Encoded</RavigerLink>
      <RavigerLink href="/login?flag">No value</RavigerLink>
      <RavigerLink href="/licenses#top">Hash</RavigerLink>
      <button onClick={() => navigate({ to: "/login", search: { next: "home" } })}>Go</button>
    </div>
  );
}
